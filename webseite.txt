function sucheAutos() {
  var typAuswahl = document.getElementById("typ").value;
  var markeAuswahl = document.getElementById("marke").value;
  var modellAuswahl = document.getElementById("Modell").value;
  var preis = document.getElementById("preis").value;

  var ergebnisse = autos.filter(auto => {
    return (
      (!typAuswahl || auto.typ === typAuswahl) &&
      (!markeAuswahl || auto.marke === markeAuswahl) &&
      (!modellAuswahl || auto.modell === modellAuswahl) &&
      (!preis || auto.preis <= parseInt(preis))
    );
  });

  var ergebnisContainer = document.getElementById("suchergebnisse");
  ergebnisContainer.innerHTML = ""; 
  if (ergebnisse.length > 0) {
    ergebnisse.forEach(auto => {
      var autoDiv = document.createElement("div");
      autoDiv.classList.add("car-card");
      autoDiv.innerHTML = `
        <h3>${auto.marke} ${auto.modell}</h3>
        <p>Typ: ${auto.typ}</p>
        <p>Preis: €${auto.preis}</p>
      `;
      ergebnisContainer.appendChild(autoDiv);
    });
  } else {
    ergebnisContainer.innerHTML = "<p>Keine Autos gefunden, die deinen Kriterien entsprechen.</p>";
  }
}

// Sparrechner
function berechnen() {
  var km = document.getElementById("km").value;
  var verbrauchE = document.getElementById("verbrauchE").value;
  var DieselPreis = document.getElementById("DieselPreis").value;
  var verbrauchB = document.getElementById("verbrauchB").value;
  var spritpreis = document.getElementById("spritpreis").value;

  var dieselKosten = (km * verbrauchE / 100) * DieselPreis;
  var benzinerKosten = (km * verbrauchB / 100) * spritpreis;

  var ergebnisDiv = document.getElementById("ergebnis");
  ergebnisDiv.innerHTML = `
    <p>Dieselkosten: €${dieselKosten.toFixed(2)}</p>
    <p>Benzinerkosten: €${benzinerKosten.toFixed(2)}</p>
  `;
}


function startQuiz() {
  var quizQuestions = [
    { question: "Wie heißt der meistverkaufte Sportwagen?", answer: "Porsche 911" },
    { question: "Was ist das älteste Auto der Welt?", answer: "Benz Patent-Motorwagen" },
    { question: "Welches Auto wird als 'Volkswagen' bezeichnet?", answer: "VW Golf" }
  ];

  var quizContainer = document.getElementById("quizContainer");
  quizContainer.innerHTML = "";

  quizQuestions.forEach((q, index) => {
    var questionDiv = document.createElement("div");
    questionDiv.innerHTML = `
      <p>${q.question}</p>
      <input type="text" id="antwort${index}">
      <button onclick="checkAnswer(${index}, '${q.answer}')">Antwort prüfen</button>
      <p id="result${index}"></p>
    `;
    quizContainer.appendChild(questionDiv);
  });
}


function checkAnswer(correctAnswer) {
  var selectedOption = document.querySelector('input[name="answer"]:checked');
  var resultDiv = document.getElementById("result");

  if (selectedOption) {
    if (selectedOption.value === correctAnswer) {
      resultDiv.innerText = "Richtig!";
      resultDiv.style.color = "green";
    } else {
      resultDiv.innerText = `Leider falsch. Die richtige Antwort ist: ${correctAnswer}`;
      resultDiv.style.color = "red";
    }
  } else {
    resultDiv.innerText = "Bitte wähle eine Antwort!";
    resultDiv.style.color = "orange";
    return;
  }

  currentQuestionIndex++;  
  setTimeout(showNextQuestion, 2000);  
}


function checkAnswer(index, correctAnswer) {
  var userAnswer = document.getElementById(`antwort${index}`).value.trim();
  var resultDiv = document.getElementById(`result${index}`);
  if (userAnswer.toLowerCase() === correctAnswer.toLowerCase()) {
    resultDiv.innerText = "Richtig!";
    resultDiv.style.color = "green";
  } else {
    resultDiv.innerText = "Leider falsch. Die richtige Antwort ist: " + correctAnswer;
    resultDiv.style.color = "red";
  }
}
